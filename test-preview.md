### Python Example

```python
# Python code with Catppuccin colors
class ColorTheme:
    def __init__(self, name):
        self.name = name
        self.colors = {
            "base": "#1e1e2e",
            "text": "#cdd6f4",
            "blue": "#89b4fa"
        }

    def apply_theme(self):
        return f"Applied {self.name} theme"

theme = ColorTheme("Catppuccin Mocha")
print(theme.apply_theme())
```
